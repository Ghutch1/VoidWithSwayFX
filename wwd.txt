# ~/.bash_profile

# ------------------------------------------------------------------------------
# 1. Source ~/.bashrc for interactive sub-shells
# ------------------------------------------------------------------------------
if [ -f ~/.bashrc ]; then
    . ~/.bashrc
fi

# ------------------------------------------------------------------------------
# 2. Wayland Environment Variables
# ------------------------------------------------------------------------------
# Default Toolkit Backends
export QT_QPA_PLATFORM=wayland
export GDK_BACKEND=wayland,x11
export ELECTRON_OZONE_PLATFORM_HINT=auto
export _JAVA_AWT_WM_NONREPARENTING=1

# Hardware Cursor (Needed on some GPUs/VM setups, safe to keep)
export WLR_NO_HARDWARE_CURSORS=1

# ------------------------------------------------------------------------------
# 3. GPU Hardware Acceleration (Uncomment your hardware profile)
# ------------------------------------------------------------------------------
# --- NVIDIA Setup ---
# export LIBVA_DRIVER_NAME=nvidia
# export __GLX_VENDOR_LIBRARY_NAME=nvidia

# --- AMD Setup ---
# export LIBVA_DRIVER_NAME=radeonsi

# --- Intel Setup ---
# export LIBVA_DRIVER_NAME=iHD

# ------------------------------------------------------------------------------
# 4. Auto-start Wayfire from TTY1
# ------------------------------------------------------------------------------
if [ -z "$WAYLAND_DISPLAY" ] && [ "$(tty)" = "/dev/tty1" ]; then
    export XDG_CURRENT_DESKTOP=Wayfire
    export XDG_SESSION_TYPE=wayland
    export XDG_SESSION_DESKTOP=Wayfire
    
    # Launch Wayfire wrapped in a D-Bus session for portal & PipeWire support
    exec dbus-run-session wayfire
fi
